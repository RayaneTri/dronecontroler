\begin{frame}{MPC control}
    \framesubtitle{Model predictive control design}

    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.52\textwidth}
            \begin{block}{Cost function minimized}
                \footnotesize
                \setlength{\abovedisplayskip}{0pt}\setlength{\belowdisplayskip}{2pt}
                \[
                    J(U) = \sum_{i=1}^{N} q\,\big(\hat{a}_{z,k+i} - r_{k+i}\big)^2
                    + \rho\, u_{k+i-1}^2
                \]
                {\scriptsize\raggedright
                $U=(u_k,\dots,u_{k+N-1})$, \ $-1\le u_{k+i}\le 1$, \ $N=20$\\
                $\hat{a}_z$: acceleration predicted by the RNN from its internal state\\
                Only $u_k$ is applied, then the horizon slides\par}
                \vspace{3pt}
                {\scriptsize\raggedright
                $J$ is minimized by gradient descent with Adam steps
                (gradients through the RNN by autodiff)\par}
            \end{block}

        \end{column}

        \begin{column}{0.44\textwidth}
            \begin{exampleblock}{Performance}
                \scriptsize
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Good overall tracking of the reference.
                    \item No phase lag: good RNN prediction.
                    \item Full amplitude not reached.
                    \item Open loop: no measurement feedback.
                \end{itemize}
            \end{exampleblock}

            \begin{alertblock}{From $a_z$ to $z$}
                \scriptsize
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item First tried: control based on $a_z$.
                    \item $a_z$ followed the reference, but $z$ diverged
                    from the request: small errors on $a_z$ accumulate
                    when integrated twice.
                    \item So the cost was moved to the height:
                \end{itemize}
                \setlength{\abovedisplayskip}{2pt}\setlength{\belowdisplayskip}{0pt}
                \[
                    J(U) = \sum_{i=1}^{N} q\,\big(\hat{z}_{k+i} - r_{k+i}\big)^2
                    + \rho\,\big(u_{k+i-1} - u_{k+i-2}\big)^2
                \]
            \end{alertblock}
        \end{column}
    \end{columns}
\end{frame}
