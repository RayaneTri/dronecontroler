\begin{frame}{Comparison}
    \framesubtitle{LQ, MPC and MPC with EKF}

    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.31\textwidth}
            \begin{block}{LQ control with Linear Model}
                \begin{itemize}
                    \item Simple and very fast.
                    \item Lack prediction.
                    \item Phase shifting.
                \end{itemize}
            \end{block}
        \end{column}

        \begin{column}{0.31\textwidth}
            \begin{block}{MPC control with RNN}
                \begin{itemize}
                    \item Prediction.
                    \item Internal state diverges from reality.
                    \item No feedback loop.
                \end{itemize}
            \end{block}
        \end{column}

        \begin{column}{0.31\textwidth}
            \begin{exampleblock}{MPC with EKF}
                \begin{itemize}
                    \item Feedback from measurements.
                    \item Position control. 
                \end{itemize}
            \end{exampleblock}
        \end{column}
    \end{columns}
\end{frame}
